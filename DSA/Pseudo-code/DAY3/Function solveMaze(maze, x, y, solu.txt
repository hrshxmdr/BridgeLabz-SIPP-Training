Function solveMaze(maze, x, y, solution)
    If x,y is destination
        solution[x][y] = 1
        return True

    If isSafe(maze, x, y)
        solution[x][y] = 1

        // Move right
        If solveMaze(maze, x+1, y, solution) return True

        // Move down
        If solveMaze(maze, x, y+1, solution) return True

        // Backtrack
        solution[x][y] = 0
        return False

    return False
End Function
