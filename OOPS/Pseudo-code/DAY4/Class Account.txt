Class Account
    Attributes: accountNumber, balance
    Function withdraw(amount)
        If balance >= amount
            balance -= amount
            Print "Withdrawn: " + amount
        Else
            Print "Insufficient Balance"

Class SavingAccount extends Account
    Attribute: minBalance = 1000
    Override Function withdraw(amount)
        If balance - amount >= minBalance
            balance -= amount
            Print "Withdrawn: " + amount
        Else
            Print "Cannot withdraw, minimum balance required"

Class CurrentAccount extends Account
    Attribute: overdraftLimit = 5000
    Override Function withdraw(amount)
        If balance + overdraftLimit >= amount
            balance -= amount
            Print "Withdrawn: " + amount
        Else
            Print "Overdraft limit exceeded"

// Example
acc1 = new SavingAccount(101, 5000)
acc1.withdraw(4500)   // Allowed

acc2 = new CurrentAccount(102, 2000)
acc2.withdraw(6000)   // Allowed (uses overdraft)
