class Item {
    String name
    double price
}

List<Item> cart = new ArrayList<Item>()

method addItem(item):
    cart.add(item)

generic method calculateTotal(List<T> items):
    total = 0
    for each item in items:
        total += item.price
    return total

// Main
addItem(new Item("Laptop", 50000))
addItem(new Item("Mouse", 500))
print("Total Price: " + calculateTotal(cart))
